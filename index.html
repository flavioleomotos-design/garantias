// GOOGLE APPS SCRIPT CODE
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var action = e.parameter.action;
  
  if (action === 'getTabs') {
    var sheets = ss.getSheets();
    var names = sheets.map(function(s) { return s.getName(); });
    return response({ status: 'success', tabs: names });
  }
  
  if (action === 'getData') {
    var sheetName = e.parameter.sheet || ss.getSheets()[0].getName();
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) return response({ status: 'error', message: 'Aba não encontrada' });
    var data = sheet.getDataRange().getValues();
    return response({ status: 'success', data: data });
  }
  
  return response({ status: 'error', message: 'Ação inválida' });
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    
    if (action === 'createTab') {
      var newSheet = ss.insertSheet(contents.tabName);
      var defaultHeaders = ['ID', 'Status', 'Cód. Barras', 'Descrição', 'Fornecedor', 'Defeito', 'Quantidade', 'Imagem', 'Observação', 'Anotações'];
      newSheet.appendRow(defaultHeaders);
      return response({ status: 'success' });
    }
    
    if (action === 'addRow') {
      var sheet = ss.getSheetByName(contents.sheet);
      if (!sheet) return response({ status: 'error', message: 'Aba não encontrada' });
      sheet.appendRow(contents.row);
      return response({ status: 'success' });
    }
    
    if (action === 'updateStatus') {
      var sheet = ss.getSheetByName(contents.sheet);
      sheet.getRange(contents.rowIndex, 2).setValue(contents.status); // Coluna 2 = Status
      return response({ status: 'success' });
    }
    
    if (action === 'deleteRow') {
      var sheet = ss.getSheetByName(contents.sheet);
      sheet.deleteRow(contents.rowIndex);
      return response({ status: 'success' });
    }

    if (action === 'addColumn') {
      var sheet = ss.getSheetByName(contents.sheet);
      var lastCol = sheet.getLastColumn();
      sheet.getRange(1, lastCol + 1).setValue(contents.columnName);
      return response({ status: 'success' });
    }
  } catch (err) {
    return response({ status: 'error', message: err.toString() });
  }
}

function response(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
